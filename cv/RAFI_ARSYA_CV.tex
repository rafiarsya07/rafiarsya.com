\section*{Selected Projects}

\project{CampusBay: Student Marketplace}{Next.js, Cloudflare Workers, D1}{Live marketplace where students
list items or post requests and close deals on WhatsApp, with instant updates and search in English,
Malay and Indonesian. Designed, built and run solo.}

\project{UMOVE: Campus Food Delivery}{React, Hono, PostgreSQL}{Food from any UM cafe, delivered by
verified UM students and paid on arrival, with a live request board, push notifications and Telegram
alerts.}

\project{Spektra: Study Reminders for UM}{FastAPI, SQLite, PWA}{Pulls SPeCTRUM deadlines, classes and
announcements automatically and reminds students on Telegram, email and push. Installable as a PWA and
Android app.}

\project{Morsel: Surplus Campus Food}{Next.js, Supabase, Tailwind}{WIX2002 Project Management course
project: campus cafes post unsold food at up to half price and students reserve it with a pickup code.}

\project{PaperMind: Local RAG Paper Analyzer}{FastAPI, ChromaDB, Ollama}{Ask questions of any PDF with
answers grounded in the text, fully offline and self-hosted.}

\project{AI Digit Recognizer: CNN From Scratch}{Python, NumPy, Flask}{Convolutional network written
entirely in NumPy, from convolutions to backpropagation and Adam: 99.1\% on the MNIST test set.}

\project{Nalar: Blog with a Custom CMS}{Node.js, Express, PostgreSQL}{Personal blog with JWT auth,
Markdown editing and scheduled publishing.}

\project{Crop Disease Detector}{TensorFlow, Keras}{Classifies 38 plant diseases from a leaf photo with
MobileNetV2 transfer learning, about 97\% accuracy, on Hugging Face Spaces.}
